# Чтобы создать столбец идентификации, воспользуйтесь предложением GENERATED ... AS IDENTITY команды CREATE TABLE, например:

function listTablesFull()
{
    $tables = listTables();
    $tables = getData('SELECT * FROM pg_class WHERE relname IN (\''.implode('\', \'', $tables).'\') ORDER BY relname');
    return $tables;
}

function primaryKey($table)
{
    $keys = getData('SELECT
    i.relname AS indexname,
    pg_get_indexdef(i.oid) AS indexdef
    FROM pg_index x
    INNER JOIN pg_class i ON i.oid = x.indexrelid
    WHERE x.indrelid = \''.$table.'\'::regclass::oid AND i.relkind = \'i\'::"char"
    AND x.indisprimary');

    preg_match('~\("([^"]+)"\)~i', $keys[0]['indexdef'], $a);

    if (!$a[1]) {
    	$fields = getFields($table);
        if (in_array('id', $fields)) {
            return 'id';
        }
    }

    return $a[1];
}

To list all indexes (including primary and unique keys):
You can use the pg_indexes system view: 

SELECT indexname, indexdef
FROM pg_indexes
WHERE tablename = 'your_table_name';


-----------------------------------------------------
To list foreign keys:
A simple way to find foreign keys is to query the information_schema.referential_constraints and information_schema.key_column_usage views: 
sql
SELECT
    kcu.column_name,
    kcu.constraint_name,
    tc.referenced_table_name,
    kcu.referenced_column_name
FROM
    information_schema.key_column_usage kcu
JOIN
    information_schema.table_constraints tc
ON
    kcu.constraint_name = tc.constraint_name
WHERE
    kcu.table_name = 'your_table_name'
    AND tc.constraint_type = 'FOREIGN KEY';
